# TalentMatch AI - backend runtime dependencies
# Install:  pip install -r requirements.txt          (add requirements-dev.txt for tests)
#
# Kept small enough for serverless hosting (Vercel's 500 MB function limit): the
# all-MiniLM-L6-v2 model runs on ONNX Runtime instead of PyTorch, with identical results.
# On Windows 11 with Smart App Control, some newer compiled wheels are blocked; the pins
# for psycopg, spacy and srsly below are builds that load correctly (see README).

# --- Web API -----------------------------------------------------------------
fastapi==0.163.1
uvicorn==0.54.0                 # local server (Vercel runs the app itself)
pydantic==2.14.3
python-multipart==0.0.34        # required by FastAPI for file uploads
python-dotenv==1.2.4            # loads backend/.env
PyJWT==2.15.1                   # signed login tokens (pure Python)

# --- Database (Supabase PostgreSQL) ------------------------------------------
SQLAlchemy==2.0.44
psycopg[binary]==3.3.5

# --- Resume parsing ----------------------------------------------------------

pymupdf==2.1 ; sys_platform == "win32"
